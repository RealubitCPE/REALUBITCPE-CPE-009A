#ifndef PALINDROME_CHECKER_H
#define PALINDROME_CHECKER_H

#include <string>

// --- 1. ARRAY-BASED STACK IMPLEMENTATION ---
class ArrayStack {
private:
    char* arr;
    int topIndex;
    int capacity;

public:
    ArrayStack(int cap = 100) {
        capacity = cap;
        arr = new char[capacity];
        topIndex = -1;
    }
    
    ~ArrayStack() {
        delete[] arr;
    }

    void push(char val) {
        if (topIndex < capacity - 1) {
            arr[++topIndex] = val;
        }
    }

    char pop() {
        if (topIndex >= 0) {
            return arr[topIndex--];
        }
        return '\0';
    }

    bool isEmpty() {
        return topIndex == -1;
    }
};

// --- 2. LINKED LIST-BASED QUEUE IMPLEMENTATION ---
struct Node {
    char data;
    Node* next;
    Node(char val) : data(val), next(nullptr) {}
};

class LinkedListQueue {
private:
    Node* frontNode;
    Node* rearNode;

public:
    LinkedListQueue() : frontNode(nullptr), rearNode(nullptr) {}
    
    ~LinkedListQueue() {
        while (!isEmpty()) {
            dequeue();
        }
    }

    void enqueue(char val) {
        Node* newNode = new Node(val);
        if (rearNode == nullptr) {
            frontNode = rearNode = newNode;
            return;
        }
        rearNode->next = newNode;
        rearNode = newNode;
    }

    char dequeue() {
        if (frontNode == nullptr) return '\0';
        Node* temp = frontNode;
        char val = temp->data;
        frontNode = frontNode->next;
        if (frontNode == nullptr) rearNode = nullptr;
        delete temp;
        return val;
    }

    bool isEmpty() {
        return frontNode == nullptr;
    }
};

// --- 3. PALINDROME ALGORITHM UTILITY ---
inline bool checkPalindrome(const std::string& str) {
    ArrayStack stack(str.length());
    LinkedListQueue queue;

    // Load data structures simultaneously
    for (char ch : str) {
        stack.push(ch);
        queue.enqueue(ch);
    }

    // Match structures (LIFO vs FIFO logic)
    while (!stack.isEmpty()) {
        if (stack.pop() != queue.dequeue()) {
            return false; // Break out immediately on mismatch
        }
    }
    return true;
}

#endif // PALINDROME_CHECKER_H

#include "PalindromeChecker.h"
#include <iostream>
#include <string>

int main() {
    // Test dataset directly from the assignment table
    std::string testCases[] = {"10201", "1003003001", "88", "0", "202"};

    for (const std::string& test : testCases) {
        std::cout << test << ": ";
        if (checkPalindrome(test)) {
            std::cout << "Is a palindrome" << std::endl;
        } else {
            std::cout << "Not a palindrome" << std::endl;
        }
    }

    return 0;
}
